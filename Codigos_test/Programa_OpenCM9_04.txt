 * Dynamixel AX-12A mediante comunicación serial USB con una aplicación
 * externa desarrollada en Python.
 *
 * FUNCIONALIDADES PRINCIPALES:
 *  - Recepción de comandos articulares desde Python.
 *  - Control de posición de motores Dynamixel.
 *  - Aplicación de límites físicos de seguridad.
 *  - Configuración de límites de torque por articulación.
 *  - Envío de telemetría en tiempo real:
 *      • Posición angular.
 *      • Temperatura de cada actuador.
 *
 * COMUNICACIÓN:
 *  - Entrada:
 *      A512B430C700...

#include <string.h>
#include <Dynamixel.h> 

// -----------------------------------------------------------------------------
// CONFIGURACIÓN DEL BUS DYNAMIXEL
// -----------------------------------------------------------------------------
#define DXL_BUS_SERIAL1 1 

// -----------------------------------------------------------------------------
// IDENTIFICADORES DE LOS ACTUADORES
// -----------------------------------------------------------------------------
#define ID_NUM1 1  
#define ID_NUM2 2  
#define ID_NUM3 3  
#define ID_NUM4 4  
#define ID_NUM5 5  
#define ID_NUM6 6  

// -----------------------------------------------------------------------------
// DIRECCIONES DE REGISTROS DYNAMIXEL AX-12A
// -----------------------------------------------------------------------------
#define TORQUE_LIMIT          34
#define PRESENT_POSITION      36 
#define PRESENT_TEMPERATURE   43 

// -----------------------------------------------------------------------------
// INSTANCIA DEL CONTROLADOR DYNAMIXEL
// -----------------------------------------------------------------------------
Dynamixel Dxl(DXL_BUS_SERIAL1);

// Variable de inicialización de postura Home
int conj = 0;

// Velocidad global
int VELOCIDAD_MOTOR = 80;

// ============================================================================
// CONFIGURACIÓN INICIAL DEL SISTEMA
// ============================================================================
void setup() {

  // Inicialización de la comunicación Dynamixel
  // Velocidad configurada a 1 Mbps
  Dxl.begin(3); 

  // Configuración de todos los actuadores en modo posición
  Dxl.jointMode(ID_NUM1);
  Dxl.jointMode(ID_NUM2);
  Dxl.jointMode(ID_NUM3);
  Dxl.jointMode(ID_NUM4);
  Dxl.jointMode(ID_NUM5);
  Dxl.jointMode(ID_NUM6);

  // ---------------------------------------------------------------------------
  // CONFIGURACIÓN DE LÍMITES DE TORQUE
  // ---------------------------------------------------------------------------
  // Se asignan diferentes límites de torque dependiendo de la carga mecánica
  // soportada por cada articulación.
  //
  // Valores:
  // 1023 -> 100% torque máximo
  //  818 -> ~80% torque máximo
  //  600 -> torque reducido
  // ---------------------------------------------------------------------------

  // Base
  Dxl.writeWord(ID_NUM1, TORQUE_LIMIT, 818);

  // Hombro
  Dxl.writeWord(ID_NUM2, TORQUE_LIMIT, 1023);

  // Codo
  Dxl.writeWord(ID_NUM3, TORQUE_LIMIT, 1023);

  // Muñeca vertical
  Dxl.writeWord(ID_NUM4, TORQUE_LIMIT, 818);

  // Rotación de muñeca
  Dxl.writeWord(ID_NUM5, TORQUE_LIMIT, 818);

  // Pinza
  Dxl.writeWord(ID_NUM6, TORQUE_LIMIT, 600);

  // Configuración de interrupción serial USB
  SerialUSB.attachInterrupt(usbInterrupt);

  // Configuración del LED integrado
  pinMode(BOARD_LED_PIN, OUTPUT);
}

// ============================================================================
// INTERRUPCIÓN DE RECEPCIÓN SERIAL USB
// ============================================================================
// Procesa los comandos recibidos desde Python.
//
// Formato esperado:
// A512B430C700D512...
//
// Cada letra representa un actuador y el número corresponde a la posición
// objetivo en unidades Dynamixel.
// ============================================================================
void usbInterrupt(byte* buffer, byte nCount) {

    char temp[64];

    // Protección contra desbordamiento de buffer
    if (nCount > 63) nCount = 63;

    // Copia del buffer recibido
    memcpy(temp, buffer, nCount);
    temp[nCount] = '\0';

    char* ptr = temp;

    // Recorrido completo de la trama serial
    while (*ptr != '\0') {

        char comando = *ptr;

        // Validación de identificador de actuador
        if (comando >= 'A' && comando <= 'F') {

            // Avanzar al valor numérico
            ptr++;

            // Conversión ASCII -> entero
            int cuenta = atoi(ptr);

            // Saturación de seguridad
            if (cuenta < 0) cuenta = 0;
            if (cuenta > 1023) cuenta = 1023;

            // -----------------------------------------------------------------
            // CONTROL INDIVIDUAL DE CADA ARTICULACIÓN
            // -----------------------------------------------------------------
            // Cada actuador posee límites físicos independientes para prevenir:
            //  - colisiones mecánicas
            //  - sobreextensión
            //  - daños estructurales
            // -----------------------------------------------------------------

            if (comando == 'A')
                moverMotor(ID_NUM1, cuenta, 170, 853);

            else if (comando == 'B')
                moverMotor(ID_NUM2, cuenta, 204, 818);

            else if (comando == 'C')
                moverMotor(ID_NUM3, cuenta, 68, 955);

            else if (comando == 'D')
                moverMotor(ID_NUM4, cuenta, 170, 853);

            else if (comando == 'E')
                moverMotor(ID_NUM5, cuenta, 130, 921);

            else if (comando == 'F')
                moverMotor(ID_NUM6, cuenta, 130, 583);

            // Saltar los dígitos ya procesados
            while (*ptr >= '0' && *ptr <= '9') {
                ptr++;
            }

        } else {

            // Ignorar caracteres inválidos
            ptr++;
        }
    }
}

// ============================================================================
// FUNCIÓN DE MOVIMIENTO CON PROTECCIÓN DE LÍMITES
// ============================================================================
// Aplica saturación de posición antes de enviar el comando al actuador.
// ============================================================================
void moverMotor(int id, int posicion, int min_lim, int max_lim) {

  // Saturación inferior
  if (posicion <= min_lim) {

    Dxl.setPosition(id, min_lim, VELOCIDAD_MOTOR);

  }
  // Saturación superior
  else if (posicion >= max_lim) {

    Dxl.setPosition(id, max_lim, VELOCIDAD_MOTOR);

  }
  // Movimiento dentro de rango permitido
  else {

    Dxl.setPosition(id, posicion, VELOCIDAD_MOTOR);
  }
}

// ============================================================================
// BUCLE PRINCIPAL
// ============================================================================
void loop() {

  // ---------------------------------------------------------------------------
  // POSTURA INICIAL HOME
  // ---------------------------------------------------------------------------
  // Se ejecuta una única vez al iniciar el sistema.
  // ---------------------------------------------------------------------------
  if (conj == 0) {

    Dxl.setPosition(ID_NUM1, 512, VELOCIDAD_MOTOR); 
    Dxl.setPosition(ID_NUM2, 512, VELOCIDAD_MOTOR);
    Dxl.setPosition(ID_NUM3, 512, VELOCIDAD_MOTOR);

    delay(1000); 

    Dxl.setPosition(ID_NUM4, 512, VELOCIDAD_MOTOR); 
    Dxl.setPosition(ID_NUM5, 716, VELOCIDAD_MOTOR);
    Dxl.setPosition(ID_NUM6, 580, VELOCIDAD_MOTOR);

    conj = 1;
  }

  // ---------------------------------------------------------------------------
  // LECTURA DE TELEMETRÍA
  // ---------------------------------------------------------------------------

  // Lectura de posición actual
  int p1 = Dxl.readWord(ID_NUM1, PRESENT_POSITION);
  int p2 = Dxl.readWord(ID_NUM2, PRESENT_POSITION);
  int p3 = Dxl.readWord(ID_NUM3, PRESENT_POSITION);
  int p4 = Dxl.readWord(ID_NUM4, PRESENT_POSITION);
  int p5 = Dxl.readWord(ID_NUM5, PRESENT_POSITION);
  int p6 = Dxl.readWord(ID_NUM6, PRESENT_POSITION);

  // Lectura de temperatura de cada actuador
  int temps[6];

  temps[0] = Dxl.readByte(ID_NUM1, PRESENT_TEMPERATURE);
  temps[1] = Dxl.readByte(ID_NUM2, PRESENT_TEMPERATURE);
  temps[2] = Dxl.readByte(ID_NUM3, PRESENT_TEMPERATURE);
  temps[3] = Dxl.readByte(ID_NUM4, PRESENT_TEMPERATURE);
  temps[4] = Dxl.readByte(ID_NUM5, PRESENT_TEMPERATURE);
  temps[5] = Dxl.readByte(ID_NUM6, PRESENT_TEMPERATURE);

  // ---------------------------------------------------------------------------
  // ENVÍO DE TELEMETRÍA HACIA PYTHON
  // ---------------------------------------------------------------------------
  // Conversión:
  // 0-1023 -> 0-300 grados
  // ---------------------------------------------------------------------------

  enviarPT(
    p1 * 300.0 / 1023.0,
    p2 * 300.0 / 1023.0,
    p3 * 300.0 / 1023.0,
    p4 * 300.0 / 1023.0,
    p5 * 300.0 / 1023.0,
    p6 * 300.0 / 1023.0,
    temps
  );

  // Frecuencia aproximada de actualización:
  // ~25 Hz
  delay(40); 
}

// ============================================================================
// ENVÍO DE POSICIÓN Y TEMPERATURA
// ============================================================================
// Formato:
// A150TA32;B140TB31;...
// ============================================================================
void enviarPT(float p1, float p2, float p3,
              float p4, float p5, float p6,
              int T[6]) {

  SerialUSB.print("A");
  SerialUSB.print(p1);
  SerialUSB.print("TA");
  SerialUSB.print(T[0]);
  SerialUSB.print(";");

  SerialUSB.print("B");
  SerialUSB.print(p2);
  SerialUSB.print("TB");
  SerialUSB.print(T[1]);
  SerialUSB.print(";");

  SerialUSB.print("C");
  SerialUSB.print(p3);
  SerialUSB.print("TC");
  SerialUSB.print(T[2]);
  SerialUSB.print(";");

  SerialUSB.print("D");
  SerialUSB.print(p4);
  SerialUSB.print("TD");
  SerialUSB.print(T[3]);
  SerialUSB.print(";");

  SerialUSB.print("E");
  SerialUSB.print(p5);
  SerialUSB.print("TE");
  SerialUSB.print(T[4]);
  SerialUSB.print(";");

  SerialUSB.print("F");
  SerialUSB.print(p6);
  SerialUSB.print("TF");
  SerialUSB.print(T[5]);
  SerialUSB.println(";");
}
